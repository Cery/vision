# 设置管理员密钥
$ADMIN = 'admin123456'

# 三个生产入口
$BASES = @(
  'https://api.visndt.com/api',
  'https://visndt.com/api',
  'https://www.visndt.com/api'
)

# 可选：同时验证 workers.dev（替换为你的子域；取消注释下一行）
# $BASES += 'https://vision-api.v-easechoice.workers.dev/api'

foreach ($base in $BASES) {
  Write-Host "=== $base ===" -ForegroundColor Cyan
  $auth = @{ 'X-Admin-Key' = $ADMIN }

  # 管理员验证：GET 与 POST
  try { $r1 = Invoke-RestMethod -Method GET  -Uri "$base/admin/verify" -Headers $auth; Write-Host "GET /admin/verify => " (ConvertTo-Json $r1) } catch { Write-Host "GET /admin/verify ERROR: $($_.Exception.Message)" -ForegroundColor Red }
  try { $r2 = Invoke-RestMethod -Method POST -Uri "$base/admin/verify" -Headers $auth -Body '{}' -ContentType 'application/json'; Write-Host "POST /admin/verify => " (ConvertTo-Json $r2) } catch { Write-Host "POST /admin/verify ERROR: $($_.Exception.Message)" -ForegroundColor Red }

  # 公共列表与详情
  try {
    $reqs = Invoke-RestMethod -Method GET -Uri "$base/requirements"
    $count = $reqs.Length
    Write-Host "GET /requirements => $count items"
    if ($count -gt 0) {
      $rid = $reqs[0].RequirementID
      $detail = Invoke-RestMethod -Method GET -Uri "$base/requirements/$rid"
      Write-Host "GET /requirements/$rid => " (ConvertTo-Json $detail)
      # 报价列表（管理员权限）
      try {
        $quotes = Invoke-RestMethod -Method GET -Uri "$base/quotes?requirement_id=$rid" -Headers $auth
        Write-Host "GET /quotes?requirement_id=$rid => count: $($quotes.Length)"
      } catch { Write-Host "GET /quotes ERROR: $($_.Exception.Message)" -ForegroundColor DarkRed }
    }
  } catch { Write-Host "GET /requirements ERROR: $($_.Exception.Message)" -ForegroundColor Red }

  # 管理列表与统计
  try {
    $admList = Invoke-RestMethod -Method GET -Uri "$base/admin/requirements?limit=5" -Headers $auth
    $n = ($admList.items | Measure-Object).Count
    Write-Host "GET /admin/requirements => $n items"
    if ($n -gt 0) {
      $firstAdmRid = $admList.items[0].requirement_id
      Write-Host "First Admin Requirement ID: $firstAdmRid"
      # 可选 PATCH 示例（默认注释，避免改动生产数据）
      # Invoke-RestMethod -Method PATCH -Uri "$base/admin/requirements/$firstAdmRid" -Headers $auth -Body '{"contact_public":1}' -ContentType 'application/json'
      # 可选 DELETE 示例（谨慎！仅在测试数据上执行）
      # Invoke-RestMethod -Method DELETE -Uri "$base/admin/requirements/$firstAdmRid" -Headers $auth
    }
  } catch { Write-Host "GET /admin/requirements ERROR: $($_.Exception.Message)" -ForegroundColor Red }

  try { $stats = Invoke-RestMethod -Method GET -Uri "$base/admin/demanders/stats" -Headers $auth; Write-Host "GET /admin/demanders/stats => " (ConvertTo-Json $stats) } catch { Write-Host "GET /admin/demanders/stats ERROR: $($_.Exception.Message)" -ForegroundColor Red }

  # 导出 CSV（展示前三行）
  try {
    $csvSup = Invoke-WebRequest -Method GET -Uri "$base/admin/suppliers/export" -Headers $auth
    Write-Host "GET /admin/suppliers/export (first lines):"
    ($csvSup.Content -split "`r?`n") | Select-Object -First 3 | ForEach-Object { Write-Host $_ }
  } catch { Write-Host "GET /admin/suppliers/export ERROR: $($_.Exception.Message)" -ForegroundColor DarkRed }

  try {
    $csvDem = Invoke-WebRequest -Method GET -Uri "$base/admin/demanders/export" -Headers $auth
    Write-Host "GET /admin/demanders/export (first lines):"
    ($csvDem.Content -split "`r?`n") | Select-Object -First 3 | ForEach-Object { Write-Host $_ }
  } catch { Write-Host "GET /admin/demanders/export ERROR: $($_.Exception.Message)" -ForegroundColor DarkRed }

  Write-Host ""
}
